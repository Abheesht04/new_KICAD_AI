find_package(Vulkan REQUIRED)
find_package(glfw3 3.3 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(IMGUI ${CMAKE_SOURCE_DIR}/third_party/imgui)
add_executable(copper-studio main.cpp studio.cpp ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_glfw.cpp ${IMGUI}/backends/imgui_impl_vulkan.cpp)
target_include_directories(copper-studio PRIVATE ${IMGUI} ${IMGUI}/backends)
target_compile_features(copper-studio PRIVATE cxx_std_17)
target_compile_definitions(copper-studio PRIVATE COPPER_ROOT="${CMAKE_SOURCE_DIR}" COPPER_CORE="$<TARGET_FILE:copper-core>")
target_link_libraries(copper-studio PRIVATE hwcore glfw Vulkan::Vulkan Threads::Threads)
add_dependencies(copper-studio copper-core)
if(MSVC)
  target_compile_options(copper-studio PRIVATE /utf-8)
endif()
